SPACE Tablewise and Skew Factor
---------------------------------

sel databasename, tablename , sum(currentperm) as "currentperm", sum(currentperm)/1024/1024/1024 Size_GB  ,
((1 - (AVG(CURRENTPERM)/MAX(CURRENTPERM)))*100) AS SKEWFACTOR
from dbc.tablesize
where databasename ='EDW01_D_LV_ODS_N_01'
and tablename  = 'SDA_EXT_CSLMTARGET'
group by databasename,tablename
order by  3 desc;

SPACE DatabaseWise and Skew Factor
-----------------------------------
SELECT Databasename (format 'X(15)') AS "Databasename"
,SUM(maxperm)/(1024*1024*1024)(FORMAT 'zzzz9.99', TITLE 'Maxperm-GB')  AS "Maxperm_GB"
,SUM(currentperm) /(1024*1024*1024)(FORMAT 'zzzz9.99', TITLE 'Currentperm-GB') AS "Currentperm_GB"
,(SUM(maxperm) - SUM(currentperm))/(1024*1024*1024)(FORMAT 'zzzz9.99', TITLE 'Free-GB') AS "Freeperm_GB"
,((SUM(currentperm))/ NULLIFZERO (SUM(maxperm)) * 100) (FORMAT 'zz9.99%', TITLE 'Percent // Used') AS "%Used",
((1 - (AVG(CURRENTPERM)/MAX(CURRENTPERM)))*100) AS SKEWFACTOR
FROM DBC.DiskSpace
WHERE
MAXPERM <> 0 
and databasename in  ('EDW01_D_LV_ODS_N_01')
--(SELECT child FROM dbc.children WHERE parent='DBC')
GROUP BY 1

SPACE DB WISE ALONG WITH SKEW DETAILS
==========================================
SELECT DATABASENAME,
SUM(CURRENTPERM)/1024/1024/1024 AS UTILIZED_SPACE_ACTUAL,
MAX(CURRENTPERM) * (HASHAMP()+1)/1024/1024/1024 AS UTILIZED_SPACE_WITH_SKEW,
UTILIZED_SPACE_WITH_SKEW - UTILIZED_SPACE_ACTUAL AS SPACE_WASTED_DUE_TO_SKEW,
SUM(MAXPERM)/1024/1024/1024 MAX_SPACE,
MAX_SPACE- UTILIZED_SPACE_WITH_SKEW AS FREE_SPACE_ACTUAL,
MAX_SPACE- UTILIZED_SPACE_ACTUAL AS FREE_SPACE,
SUM(CURRENTPERM)/ NULLIFZERO (SUM(MAXPERM)) *100 (FORMAT 'ZZ9.99%') AS PERCENTAGE_USED 
FROM DBC.DISKSPACE
WHERE DATABASENAME IN 
('SPOOL_RESERVE', 'TMP','EDW01_D_LV_ODS_N_01')
GROUP BY DATABASENAME
ORDER BY 2

DORMANT USER FINDING

create volatile table tmplogonoff as (
sel username, MechanismName, max(logondate)  mxlogondate from (
sel username,MechanismName, max(logondate) logondate from  pdcrinfo.logonoff_hst t1 
group by 1,2
UNION
sel username,MechanismName, max(logondate) logondate from  dbc.logonoff t1 
group by 1,2
) T
group by 1,2
) with data
primary index(username)
on commit preserve rows;

collect stats on tmplogonoff index(username);
collect stats on tmplogonoff column(username,MechanismName);
